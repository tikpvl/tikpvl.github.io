Чтобы данные с телефонов секретарей ОИК (Сайт №1) мгновенно собирались на вашем компьютере ТИК (Сайт №2), создайте Google Таблицу, откройте Расширения → Apps Script, вставьте код ниже и нажмите Развернуть → Новое развертывание (Веб-приложение, доступ: Все).

Код сервера Google Apps Script (Code.gs): 
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var sheet = getOrCreateSheet();
    var d = JSON.parse(e.postData.contents);

    // Удаление строки по ID (когда запись удалили в интерфейсе ТИК)
    if (d.action === 'delete') {
      var data = sheet.getDataRange().getValues();
      for (var i = data.length - 1; i >= 1; i--) {
        if (String(data[i][0]) === String(d.id)) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({ status: 'ok', deleted: d.id }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // Добавление или обновление строки (защита от дубликатов при повторной отправке)
    var data = sheet.getDataRange().getValues();
    var existingRow = -1;
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][0]) === String(d.id) ||
         (String(data[i][1]) === String(d.date) && String(data[i][4]) === String(d.district) && String(data[i][5]) === String(d.oikNumber) && String(data[i][3]) === String(d.level))) {
        existingRow = i + 1;
        break;
      }
    }

    var rowValues = [
      d.id, d.date, d.region, d.level, d.district, d.oikNumber,
      d.secretary, d.totalCandidates, d.selfNominated,
      (d.parties && d.parties['AMANAT']) || 0,
      (d.parties && d.parties['Ақ жол']) || 0,
      (d.parties && d.parties['Ауыл']) || 0,
      (d.parties && d.parties['НПК']) || 0,
      (d.parties && d.parties['Respublica']) || 0,
      (d.parties && d.parties['ОСДП']) || 0,
      (d.parties && d.parties['Байтақ']) || 0,
      d.compliantCount, d.pendingCount, d.nonCompliantCount,
      d.complianceStatus, d.notes, (d.submittedAt ? (d.date + ' ' + d.submittedAt) : new Date().toLocaleString('ru-RU'))
    ];

    if (existingRow > 0) {
      sheet.getRange(existingRow, 1, 1, rowValues.length).setValues([rowValues]);
    } else {
      sheet.appendRow(rowValues);
    }

    return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(5000);
  try {
    var sheet = getOrCreateSheet();
    var values = sheet.getDataRange().getValues();
    var records = [];

    // values[0] - заголовки, данные начинаются со строки 1
    for (var i = 1; i < values.length; i++) {
      var row = values[i];
      if (!row[0] && !row[5]) continue; // Пропуск пустых строк
      records.push({
        id: String(row[0] || ('rec-' + i)),
        date: String(row[1] || ''),
        region: String(row[2] || 'Павлодарская область'),
        level: String(row[3] || 'Городской / Районный маслихат'),
        district: String(row[4] || ''),
        oikNumber: parseInt(row[5], 10) || 0,
        secretary: String(row[6] || ''),
        totalCandidates: parseInt(row[7], 10) || 0,
        selfNominated: parseInt(row[8], 10) || 0,
        parties: {
          'AMANAT': parseInt(row[9], 10) || 0,
          'Ақ жол': parseInt(row[10], 10) || 0,
          'Ауыл': parseInt(row[11], 10) || 0,
          'НПК': parseInt(row[12], 10) || 0,
          'Respublica': parseInt(row[13], 10) || 0,
          'ОСДП': parseInt(row[14], 10) || 0,
          'Байтақ': parseInt(row[15], 10) || 0
        },
        compliantCount: parseInt(row[16], 10) || 0,
        pendingCount: parseInt(row[17], 10) || 0,
        nonCompliantCount: parseInt(row[18], 10) || 0,
        complianceStatus: String(row[19] || 'Полное соответствие'),
        notes: String(row[20] || ''),
        submittedAt: String(row[21] || '')
      });
    }

    var output = JSON.stringify({ status: 'ok', records: records, rows: values });
    if (e && e.parameter && e.parameter.callback) {
      return ContentService.createTextOutput(e.parameter.callback + '(' + output + ')')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(output)
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateSheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('Сводка_ОИК') || ss.insertSheet('Сводка_ОИК');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      'ID', 'Дата', 'Область', 'Уровень маслихата', 'Город / Район', 'Номер ОИК',
      'ФИО Секретаря', 'Всего кандидатов', 'Самовыдвижение',
      'AMANAT', 'Ақ жол', 'Ауыл', 'НПК', 'Respublica', 'ОСДП', 'Байтақ',
      'Соответствуют', 'На проверке', 'Не соответствуют', 'Статус соответствия', 'Примечание', 'Время подачи'
    ]);
  }
  return sheet;
}
